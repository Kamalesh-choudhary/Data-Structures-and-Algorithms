t = int(input())
for _ in range(t):
    n,k = map(int,input().split())
    strips = [input().strip() for _ in range(k)]
    allowed = [set() for i in range(n)]
    for i in range(n):
        for s in strips:
            allowed[i].add(s[i])
    divisors = []
    for d in range(1,int(n**0.5)+1):
        if n%d ==0:
            divisors.append(d)
            if d!= n//d:
                divisors.append(n//d)
    divisors.sort()
    answer = None
    for d in divisors:
        ok = True
        chosen = [''] * d   
 
        for r in range(d):
            common = allowed[r].copy()
            pos = r + d
            while pos < n:
                common &= allowed[pos]
                if not common:
                    ok = False
                    break
                pos += d
            if not ok:
                break
 
            chosen[r] = next(iter(common))
 
        if ok:
            res = [''] * n
            for i in range(n):
                res[i] = chosen[i % d]
            answer = ''.join(res)
            break
 
    print(answer)
