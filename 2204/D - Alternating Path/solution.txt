from collections import deque
def dfs(start,graph,visited):
    global count0,count1
    q = deque()
    q.append(start)
    bip = True
    visited[start] = 0
    count0 += 1    
    while q:
        u = q.pop()
        for v in graph[u]:
            if visited[v] == -1:
                visited[v] = not visited[u]
                q.append(v)
                if visited[v] == 0:
                    count0 += 1
                else:
                    count1 += 1
            elif visited[v] == visited[u]:
                bip = False
            
    return bip
 
 
t = int(input())
for _ in range(t):
    n,m = map(int,input().split())
    graph = [[] for _ in range(n+1)]
    for _ in range(m):
        u,v = map(int,input().split())
        graph[u].append(v)
        graph[v].append(u)
    visited = [-1]*(n+1)
    ans = 0
    for i in range(1,n+1):
        if visited[i] != -1:
            continue
        count0 = 0
        count1 = 0
        if dfs(i,graph,visited):
            ans += max(count1,count0)
    print(ans)
