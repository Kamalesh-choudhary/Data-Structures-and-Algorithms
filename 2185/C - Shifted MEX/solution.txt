t = int(input())
for _ in range(t):
    n = int(input())
    ar = list(map(int,input().split()))
    unique = sorted(list(set(ar)))
    ans = 0
    cur_continuous = 0
 
    for i in range(len(unique)):
        if i>0 and unique[i] == unique[i-1]+1:
            cur_continuous += 1
        else:
            cur_continuous = 1
 
        if cur_continuous > ans:
            ans = cur_continuous
    if n==0:
        print(0)
    else:
        print(ans)
